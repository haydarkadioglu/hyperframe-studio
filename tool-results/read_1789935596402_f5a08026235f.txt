   436→   436→function ProductMode() {
   437→   437→  const wizard = useApp((s) => s.wizard);
   438→   438→  const setWizard = useApp((s) => s.setWizard);
   439→   439→  const fileRef = React.useRef<HTMLInputElement>(null);
   440→   440→
   441→   441→  const [uploading, setUploading] = React.useState(false);
   442→   442→  const [analyzing, setAnalyzing] = React.useState(false);
   443→   443→  const [analysis, setAnalysis] = React.useState<AnalyzeProductResponse | null>(null);
   444→   444→
   445→   445→  const onPick = async (files: FileList | null) => {
   446→   446→    if (!files || !files.length) return;
   447→   447→    setUploading(true);
   448→   448→    try {
   449→   449→      const collected: string[] = [];
   450→   450→      for (const file of Array.from(files).slice(0, 8)) {
   451→   451→        const dataUrl = await fileToDataUrl(file);
   452→   452→        const { url } = await uploadImage(dataUrl, file.name);
   453→   453→        // For product analysis + render, we need both the persisted URL (for display)
   454→   454→        // and the data URL (for VLM). We persist to /assets/ via upload and keep dataUrl
   455→   455→        // for sending to render. Store BOTH: the dataUrl so it can be sent to render API.
   456→   456→        // The display layer uses the persisted url OR the dataUrl (both work in <img>).
   457→   457→        collected.push(dataUrl);
   458→   458→      }
   459→   459→      setWizard({ productImages: [...wizard.productImages, ...collected] });
   460→   460→      toast.success(`${collected.length} görsel yüklendi`);
   461→   461→    } catch (e: any) {
   462→   462→      toast.error("Yükleme hatası", { description: e?.message });
   463→   463→    } finally {
   464→   464→      setUploading(false);
   465→   465→      if (fileRef.current) fileRef.current.value = "";
   466→   466→    }
   467→   467→  };
   468→   468→
   469→   469→  const removeImage = (idx: number) => {
   470→   470→    setWizard({
