   432→// ---------- KPI Card with count-up ----------
   433→function KpiCard({
   434→  icon: Icon,
   435→  gradient,
   436→  value,
   437→  label,
   438→  subtitle,
   439→  formatDurationValue,
   440→}: {
   441→  icon: React.ComponentType<{ className?: string }>;
   442→  gradient: string;
   443→  value: number;
   444→  label: string;
   445→  subtitle: string;
   446→  formatDurationValue?: boolean;
   447→}) {
   448→  const mv = useMotionValue(0);
   449→  const [display, setDisplay] = React.useState("0");
   450→  const displayRef = React.useRef<(s: string) => void>((s) => setDisplay(s));
   451→
   452→  React.useEffect(() => {
   453→    const target = value;
   454→    if (formatDurationValue) {
   455→      // animate the second count
   456→      const controls = animate(mv, target, {
   457→        duration: 0.8,
   458→        ease: "easeOut",
   459→        onUpdate: (v) => {
   460→          const m = Math.floor(v / 60);
   461→          const s = Math.floor(v % 60);
   462→          displayRef.current(`${m}d ${s}s`);
   463→        },
   464→      });
   465→      return () => controls.stop();
   466→    }
   467→    const controls = animate(mv, target, {
   468→      duration: 0.8,
   469→      ease: "easeOut",
   470→      onUpdate: (v) => {
   471→        displayRef.current(String(Math.round(v)));
   472→      },
   473→    });
   474→    return () => controls.stop();
   475→  }, [value, formatDurationValue, mv]);
   476→
   477→  return (
   478→    <motion.div variants={item}>
   479→      <Card className="glass card-glow shine-on-hover card-hover-lift h-full overflow-hidden group">
   480→        <CardContent className="p-5">
   481→          <div className="flex items-start justify-between">
   482→            <div
   483→              className={cn(
   484→                "relative grid size-11 place-items-center rounded-xl accent-gradient text-white shadow-md"
   485→              )}
   486→            >
   487→              <Icon className="size-5" />
   488→              {/* Rotating gradient ring on hover */}
   489→              <span className="absolute -inset-1 rounded-xl ring-1 ring-fuchsia-500/30 opacity-0 group-hover:opacity-100 group-hover:rotate-45 transition-all duration-500 pointer-events-none" />
   490→            </div>
   491→            <ArrowRight className="size-4 text-muted-foreground/40 group-hover:text-fuchsia-500 group-hover:translate-x-0.5 transition-all" />
   492→          </div>
   493→          <div className="mt-4">
   494→            <p className="text-3xl font-bold tracking-tight tabular-nums">
   495→              {display}
   496→            </p>
   497→            <p className="text-sm font-medium mt-0.5">{label}</p>
   498→            <p className="text-[11px] text-muted-foreground">{subtitle}</p>
   499→          </div>
   500→        </CardContent>
   501→      </Card>
   502→    </motion.div>
   503→  );
   504→}

   506→// ---------- Donut pie chart ----------
   507→function DistributionPie({
   508→  data,
   509→}: {
   510→  data: { id: string; label: string; emoji: string; count: number }[];
   511→}) {
   512→  const total = data.reduce((a, b) => a + b.count, 0);
   513→  if (total === 0) {
   514→    return <EmptyChart label="Henüz veri yok" />;
   515→  }
   516→  const chartData = data.map((d) => ({
   517→    name: `${d.emoji} ${d.label}`,
   518→    value: d.count,
   519→  }));
   520→  return (
   521→    <div className="flex flex-col sm:flex-row items-center gap-4">
   522→      <div className="h-[200px] w-full sm:w-1/2">
   523→        <ResponsiveContainer width="100%" height="100%">
   524→          <PieChart>
   525→            <Pie
   526→              data={chartData}
   527→              dataKey="value"
   528→              nameKey="name"
   529→              cx="50%"
   530→              cy="50%"
   531→              innerRadius={55}
   532→              outerRadius={85}
   533→              paddingAngle={3}
   534→              stroke="rgba(0,0,0,0.2)"
   535→              strokeWidth={1}
   536→            >
   537→              {chartData.map((_, i) => (
   538→                <Cell key={i} fill={CHART_COLORS[i % CHART_COLORS.length]} />
   539→              ))}
   540→            </Pie>
   541→            <Tooltip
   542→              contentStyle={{
   543→                backgroundColor: "rgba(20,20,22,0.92)",
   544→                border: "1px solid rgba(217,70,239,0.3)",
   545→                borderRadius: 12,
   546→                color: "#fff",
   547→                fontSize: 12,
   548→              }}
   549→              formatter={(v: number, n: string) => [`${v} video`, n]}
   550→            />
   551→          </PieChart>
   552→        </ResponsiveContainer>
   553→      </div>
   554→      <ul className="flex-1 w-full sm:w-1/2 space-y-1.5">
   555→        {data.map((d, i) => {
   556→          const pct = total > 0 ? Math.round((d.count / total) * 100) : 0;
   557→          return (
   558→            <li
   559→              key={d.id}
   560→              className="flex items-center justify-between gap-2 text-sm py-1"
   561→            >
   562→              <span className="flex items-center gap-2 min-w-0">
   563→                <span
   564→                  className="size-2.5 rounded-full shrink-0"
   565→                  style={{ background: CHART_COLORS[i % CHART_COLORS.length] }}
   566→                />
   567→                <span className="truncate">
   568→                  {d.emoji} {d.label}
   569→                </span>
   570→              </span>
   571→              <span className="flex items-center gap-2 shrink-0">
   572→                <span className="text-xs text-muted-foreground tabular-nums">
   573→                  %{pct}
   574→                </span>
   575→                <span className="font-semibold tabular-nums">{d.count}</span>
   576→              </span>

   670→// ---------- Provider usage card ----------
   671→function ProviderUsageCard({
   672→  title,
   673→  icon: Icon,
   674→  accent,
   675→  data,
   676→  resolver,
   677→}: {
   678→  title: string;
   679→  icon: React.ComponentType<{ className?: string }>;
   680→  accent: "violet" | "fuchsia";
   681→  data: { id: string; count: number }[];
   682→  resolver: (id: string) => string;
   683→}) {
   684→  const total = data.reduce((a, b) => a + b.count, 0);
   685→  const accentClass =
   686→    accent === "violet"
   687→      ? "from-violet-500 to-fuchsia-500 text-violet-300"
   688→      : "from-fuchsia-500 to-pink-500 text-fuchsia-300";
   689→  return (
   690→    <Card className="glass card-glow h-full">
   691→      <CardHeader>
   692→        <div className="flex items-center gap-2">
   693→          <span
   694→            className={cn(
   695→              "grid size-8 place-items-center rounded-lg bg-gradient-to-br text-white shadow-md",
   696→              accentClass
   697→            )}
   698→          >
   699→            <Icon className="size-4" />
   700→          </span>
   701→          <div>
   702→            <CardTitle className="text-base">{title}</CardTitle>
   703→            <p className="text-xs text-muted-foreground">
   704→              Toplam {total} kullanım
   705→            </p>
   706→          </div>
   707→        </div>
   708→      </CardHeader>
   709→      <CardContent className="space-y-2">
   710→        {data.length === 0 ? (
   711→          <p className="text-sm text-muted-foreground py-4 text-center">
   712→            Henüz veri yok
   713→          </p>
   714→        ) : (
   715→          data.map((d, i) => {
   716→            const name = resolver(d.id);
   717→            const pct = total > 0 ? Math.round((d.count / total) * 100) : 0;
   718→            return (
   719→              <motion.div
   720→                key={d.id + i}
   721→                initial={{ opacity: 0, x: -8 }}
   722→                animate={{ opacity: 1, x: 0 }}
   723→                transition={{ delay: i * 0.04 }}
   724→                className="flex items-center gap-3"
   725→              >
   726→                <div className="min-w-0 flex-1">
   727→                  <div className="flex items-center justify-between text-sm mb-1">
   728→                    <span className="truncate font-medium">{name}</span>
   729→                    <span className="text-xs text-muted-foreground tabular-nums">
   730→                      {d.count} · %{pct}
   731→                    </span>
   732→                  </div>
   733→                  <div className="h-1.5 w-full rounded-full bg-muted overflow-hidden">
   734→                    <div
   735→                      className={cn(
   736→                        "h-full rounded-full bg-gradient-to-r",
   737→                        accentClass
   738→                      )}
   739→                      style={{ width: `${pct}%` }}
   740→                    />
   741→                  </div>
   742→                </div>
   743→                <Badge
   744→                  variant="outline"
   745→                  className={cn("shrink-0", accentClass)}
   746→                >
   747→                  {d.count}
   748→                </Badge>
   749→              </motion.div>
   750→            );
   751→          })
   752→        )}
   753→      </CardContent>
   754→    </Card>
   755→  );
   756→}
