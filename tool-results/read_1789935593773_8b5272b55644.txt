    78→export function CreateView() {
    79→  const wizard = useApp((s) => s.wizard);
    80→  const setWizard = useApp((s) => s.setWizard);

    83→

    85→

    92→      if (wizard.mode === "youtube") return wizard.topic.trim().length >= 3;
    93→      if (wizard.mode === "product") return wizard.productImages.length > 0;
    94→      if (wizard.mode === "script") return wizard.customScript.trim().length >= 30;
    95→    }
    96→    return true;
    97→  }, [step, wizard]);
    98→
    99→  return (
   100→    <div className="space-y-6">
   101→      <header className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   102→        <div>
   103→          <p className="text-xs uppercase tracking-[0.2em] text-fuchsia-500 font-semibold">
   104→            Yeni Video
   105→          </p>
   106→          <h1 className="mt-1 text-2xl sm:text-3xl font-bold tracking-tight">
   107→            Video Oluştur
   108→          </h1>
   109→          <p className="text-sm text-muted-foreground mt-1">
   110→            4 adımda videonuzu üretin.
   111→          </p>
   112→        </div>
   113→        <Button
   114→          variant="ghost"
   115→          size="sm"
   116→          onClick={() => {
   117→            resetWizard();
   118→            go("home");
   119→          }}
   120→        >
   121→          Sıfırla
   122→        </Button>
   123→      </header>
   124→
   125→      {/* Step indicator */}
   126→      <ol className="flex flex-wrap gap-2">
   127→        {STEPS.map((label, i) => {
   128→          const active = i === step;
   129→          const done = i < step;
   130→          return (
   131→            <li key={label}>
   132→              <button
   133→                onClick={() => i < step && setStep(i)}
   134→                disabled={i > step}
   135→                className={cn(
   136→                  "inline-flex items-center gap-2 rounded-full border px-3 py-1.5 text-xs font-medium transition-colors min-h-[36px]",
   137→                  active
   138→                    ? "border-fuchsia-500 bg-fuchsia-500/10 text-foreground"
   139→                    : done
   140→                    ? "border-emerald-500/40 bg-emerald-500/10 text-emerald-500 hover:bg-emerald-500/15"
   141→                    : "border-border text-muted-foreground cursor-not-allowed"
   142→                )}
   143→              >
   144→                <span
   145→                  className={cn(
   146→                    "grid size-5 place-items-center rounded-full text-[10px]",
   147→                    active
   148→                      ? "bg-gradient-to-br from-violet-500 to-fuchsia-500 text-white"
   149→                      : done
   150→                      ? "bg-emerald-500 text-white"
   151→                      : "bg-muted text-muted-foreground"
   152→                  )}
   153→                >
   154→                  {done ? <Check className="size-3" /> : i + 1}
   155→                </span>
   156→                {label}
   157→              </button>
   158→            </li>
   159→          );
   160→        })}
   161→      </ol>
   162→
   163→      <AnimatePresence mode="wait">
   164→        <motion.div
   165→          key={step}
   166→          initial={{ opacity: 0, y: 14 }}
   167→          animate={{ opacity: 1, y: 0 }}
   168→          exit={{ opacity: 0, y: -10 }}
   169→          transition={{ duration: 0.3 }}
   170→        >
   171→          {step === 0 && <StepMode />}
   172→          {step === 1 && <StepContent />}
   173→          {step === 2 && <StepConfig />}
   174→          {step === 3 && <StepSummary />}
   175→        </motion.div>
   176→      </AnimatePresence>
   177→
   178→      {/* Nav buttons */}
   179→      <div className="flex items-center justify-between gap-3 pt-4 border-t border-border">
   180→        <Button
   181→          variant="ghost"
   182→          onClick={() => {
   183→            if (step === 0) go("home");
   184→            else setStep(step - 1);
   185→          }}
   186→          className="min-h-[44px]"
   187→        >
   188→          <ArrowLeft className="size-4" />
   189→          {step === 0 ? "İptal" : "Geri"}
   190→        </Button>
   191→        {step < 3 && (
   192→          <Button
   193→            onClick={() => canNext && setStep(step + 1)}
   194→            disabled={!canNext}
   195→            className="bg-gradient-to-r from-violet-500 via-fuchsia-500 to-pink-500 text-white border-0 min-h-[44px]"
   196→          >

   436→function ProductMode() {
   437→  const wizard = useApp((s) => s.wizard);
   438→  const setWizard = useApp((s) => s.setWizard);
   439→  const fileRef = React.useRef<HTMLInputElement>(null);
   440→
   441→  const [uploading, setUploading] = React.useState(false);
   442→  const [analyzing, setAnalyzing] = React.useState(false);
   443→  const [analysis, setAnalysis] = React.useState<AnalyzeProductResponse | null>(null);
   444→
   445→  const onPick = async (files: FileList | null) => {
   446→    if (!files || !files.length) return;
   447→    setUploading(true);
   448→    try {
   449→      const collected: string[] = [];
   450→      for (const file of Array.from(files).slice(0, 8)) {
   451→        const dataUrl = await fileToDataUrl(file);
   452→        const { url } = await uploadImage(dataUrl, file.name);
   453→        // For product analysis + render, we need both the persisted URL (for display)
   454→        // and the data URL (for VLM). We persist to /assets/ via upload and keep dataUrl
   455→        // for sending to render. Store BOTH: the dataUrl so it can be sent to render API.
   456→        // The display layer uses the persisted url OR the dataUrl (both work in <img>).
   457→        collected.push(dataUrl);
   458→      }
   459→      setWizard({ productImages: [...wizard.productImages, ...collected] });
   460→      toast.success(`${collected.length} görsel yüklendi`);
   461→    } catch (e: any) {
   462→      toast.error("Yükleme hatası", { description: e?.message });
   463→    } finally {
   464→      setUploading(false);
   465→      if (fileRef.current) fileRef.current.value = "";
   466→    }
   467→  };
   468→
   469→  const removeImage = (idx: number) => {
   470→    setWizard({
   471→      productImages: wizard.productImages.filter((_, i) => i !== idx),
   472→    });
   473→  };
   474→
   475→  const runAnalysis = async () => {
   476→    const first = wizard.productImages[0];
   477→    if (!first) return;
   478→    setAnalyzing(true);
   479→    setAnalysis(null);
   480→    try {
   481→      const res = await analyzeProduct({
   482→        image: first,
   483→        language: wizard.language,
   484→      });
   485→      setAnalysis(res);
   486→      toast.success("Ürün analizi tamamlandı");
   487→    } catch (e: any) {
   488→      toast.error("Analiz hatası", { description: e?.message });
   489→    } finally {
   490→      setAnalyzing(false);
   491→    }
   492→  };
   493→
   494→  return (
   495→    <div className="space-y-4">
   496→      <Card>
   497→        <CardHeader>
   498→          <CardTitle className="flex items-center gap-2">
   499→            <span className="grid size-7 place-items-center rounded-lg bg-amber-500/20 text-amber-500">
   500→              <ImagePlus className="size-4" />
   501→            </span>
   502→            Ürün Görselleri
   503→          </CardTitle>
   504→          <CardDescription>
   505→            Ürün fotoğraflarınızı yükleyin. AI (VLM) ile analiz edilecek ve
   506→            tanıtım videosu oluşturulacak.
   507→          </CardDescription>
   508→        </CardHeader>
   509→        <CardContent className="space-y-4">
   510→          <div
   511→            onDragOver={(e) => {
   512→              e.preventDefault();
   513→              e.dataTransfer.dropEffect = "copy";
   514→            }}
   515→            onDrop={(e) => {
   516→              e.preventDefault();
   517→              onPick(e.dataTransfer.files);
   518→            }}
   519→            onClick={() => fileRef.current?.click()}
   520→            className="cursor-pointer rounded-2xl border-2 border-dashed border-border hover:border-fuchsia-500/50 hover:bg-fuchsia-500/5 transition-colors p-8 text-center"
   521→          >
   522→            <input
   523→              ref={fileRef}
   524→              type="file"
   525→              accept="image/*"
   526→              multiple
   527→              className="hidden"
   528→              onChange={(e) => onPick(e.target.files)}
   529→            />
   530→            <div className="mx-auto grid size-12 place-items-center rounded-full bg-muted text-muted-foreground mb-3">
   531→              {uploading ? (
   532→                <Loader2 className="size-5 animate-spin text-fuchsia-500" />
   533→              ) : (
   534→                <Upload className="size-5" />
   535→              )}
   536→            </div>
   537→            <p className="text-sm font-medium">
   538→              {uploading
   539→                ? "Yükleniyor..."
   540→                : "Sürükleyip bırakın veya tıklayın"}
   541→            </p>
   542→            <p className="text-xs text-muted-foreground mt-1">
   543→              PNG, JPG, WEBP · en fazla 8 görsel
   544→            </p>
   545→          </div>
   546→
   547→          {wizard.productImages.length > 0 && (
   548→            <div className="grid grid-cols-3 sm:grid-cols-4 gap-3">
   549→              {wizard.productImages.map((img, i) => (
   550→                <div
   551→                  key={i}
   552→                  className="relative aspect-square overflow-hidden rounded-lg border border-border group"
   553→                >
   554→                  <img
   555→                    src={img}
   556→                    alt={`Ürün ${i + 1}`}
   557→                    className="size-full object-cover"
   558→                  />
   559→                  <button
   560→                    onClick={() => removeImage(i)}
   561→                    className="absolute top-1 right-1 grid size-6 place-items-center rounded-full bg-black/70 text-white opacity-0 group-hover:opacity-100 transition-opacity"
   562→                    aria-label="Kaldır"
   563→                  >
   564→                    <Trash2 className="size-3" />
   565→                  </button>
   566→                </div>
   567→              ))}
   568→            </div>
   569→          )}
