   429→// ---------- KPI Card with count-up ----------
   430→function KpiCard({
   431→  icon: Icon,
   432→  gradient,
   433→  value,
   434→  label,
   435→  subtitle,
   436→  formatDurationValue,
   437→}: {
   438→  icon: React.ComponentType<{ className?: string }>;
   439→  gradient: string;
   440→  value: number;
   441→  label: string;
   442→  subtitle: string;
   443→  formatDurationValue?: boolean;
   444→}) {
   445→  const mv = useMotionValue(0);
   446→  const [display, setDisplay] = React.useState("0");
   447→  const displayRef = React.useRef<(s: string) => void>((s) => setDisplay(s));
   448→
   449→  React.useEffect(() => {
   450→    const target = value;
   451→    if (formatDurationValue) {
   452→      // animate the second count
   453→      const controls = animate(mv, target, {
   454→        duration: 0.8,
   455→        ease: "easeOut",
   456→        onUpdate: (v) => {
   457→          const m = Math.floor(v / 60);
   458→          const s = Math.floor(v % 60);
   459→          displayRef.current(`${m}d ${s}s`);
   460→        },
   461→      });
   462→      return () => controls.stop();
   463→    }
   464→    const controls = animate(mv, target, {
   465→      duration: 0.8,
   466→      ease: "easeOut",
   467→      onUpdate: (v) => {
   468→        displayRef.current(String(Math.round(v)));
   469→      },
   470→    });
   471→    return () => controls.stop();
   472→  }, [value, formatDurationValue, mv]);
   473→
   474→  return (
   475→    <motion.div variants={item}>
   476→      <Card className="glass card-glow shine-on-hover card-hover-lift h-full overflow-hidden group">
   477→        <CardContent className="p-5">
   478→          <div className="flex items-start justify-between">
   479→            <div
   480→              className={cn(
   481→                "relative grid size-11 place-items-center rounded-xl bg-gradient-to-br text-white shadow-md",
   482→                gradient
   483→              )}
   484→            >
   485→              <Icon className="size-5" />
   486→              {/* Rotating gradient ring on hover */}
   487→              <span className="absolute -inset-1 rounded-xl ring-1 ring-fuchsia-500/30 opacity-0 group-hover:opacity-100 group-hover:rotate-45 transition-all duration-500 pointer-events-none" />
   488→            </div>
   489→            <ArrowRight className="size-4 text-muted-foreground/40 group-hover:text-fuchsia-500 group-hover:translate-x-0.5 transition-all" />
   490→          </div>
   491→          <div className="mt-4">
   492→            <p className="text-3xl font-bold tracking-tight tabular-nums">
   493→              {display}
   494→            </p>
   495→            <p className="text-sm font-medium mt-0.5">{label}</p>
   496→            <p className="text-[11px] text-muted-foreground">{subtitle}</p>
   497→          </div>
   498→        </CardContent>
   499→      </Card>
   500→    </motion.div>
   501→  );
   502→}

   504→// ---------- Donut pie chart ----------
   505→function DistributionPie({
   506→  data,
   507→}: {
   508→  data: { id: string; label: string; emoji: string; count: number }[];
   509→}) {
   510→  const total = data.reduce((a, b) => a + b.count, 0);
   511→  if (total === 0) {
   512→    return <EmptyChart label="Henüz veri yok" />;
   513→  }
   514→  const chartData = data.map((d) => ({
   515→    name: `${d.emoji} ${d.label}`,
   516→    value: d.count,
   517→  }));
   518→  return (
   519→    <div className="flex flex-col sm:flex-row items-center gap-4">
   520→      <div className="h-[200px] w-full sm:w-1/2">
   521→        <ResponsiveContainer width="100%" height="100%">
   522→          <PieChart>
   523→            <Pie
   524→              data={chartData}
   525→              dataKey="value"
   526→              nameKey="name"
   527→              cx="50%"
   528→              cy="50%"
   529→              innerRadius={55}
   530→              outerRadius={85}
   531→              paddingAngle={3}
   532→              stroke="rgba(0,0,0,0.2)"
   533→              strokeWidth={1}
   534→            >
   535→              {chartData.map((_, i) => (
   536→                <Cell key={i} fill={CHART_COLORS[i % CHART_COLORS.length]} />
   537→              ))}
   538→            </Pie>
   539→            <Tooltip
   540→              contentStyle={{
   541→                backgroundColor: "rgba(20,20,22,0.92)",
   542→                border: "1px solid rgba(217,70,239,0.3)",
   543→                borderRadius: 12,
   544→                color: "#fff",
   545→                fontSize: 12,
   546→              }}
   547→              formatter={(v: number, n: string) => [`${v} video`, n]}
   548→            />
   549→          </PieChart>
   550→        </ResponsiveContainer>
   551→      </div>
   552→      <ul className="flex-1 w-full sm:w-1/2 space-y-1.5">
   553→        {data.map((d, i) => {
   554→          const pct = total > 0 ? Math.round((d.count / total) * 100) : 0;
   555→          return (
   556→            <li
   557→              key={d.id}
   558→              className="flex items-center justify-between gap-2 text-sm py-1"
   559→            >
   560→              <span className="flex items-center gap-2 min-w-0">
   561→                <span
   562→                  className="size-2.5 rounded-full shrink-0"
   563→                  style={{ background: CHART_COLORS[i % CHART_COLORS.length] }}
   564→                />
   565→                <span className="truncate">
   566→                  {d.emoji} {d.label}
   567→                </span>
   568→              </span>
   569→              <span className="flex items-center gap-2 shrink-0">
   570→                <span className="text-xs text-muted-foreground tabular-nums">
   571→                  %{pct}
   572→                </span>
   573→                <span className="font-semibold tabular-nums">{d.count}</span>
   574→              </span>
   575→            </li>
   576→          );
   577→        })}
   578→      </ul>
   579→    </div>
   580→  );
   581→}

   584→function LanguageBar({
   585→  data,
   586→}: {
   587→  data: { code: string; name: string; flag: string; count: number }[];
   588→}) {
   589→  const total = data.reduce((a, b) => a + b.count, 0);
   590→  if (total === 0) {
   591→    return <EmptyChart label="Henüz veri yok" />;

   593→  const chartData = data.map((d) => ({
   594→    name: `${d.flag} ${d.code.toUpperCase()}`,
   595→    fullName: d.name,
   596→    value: d.count,
   597→  }));
   598→  return (
   599→    <div className="h-[260px] w-full">

   602→          data={chartData}
   603→          layout="vertical"
   604→          margin={{ top: 5, right: 30, left: 10, bottom: 5 }}

   607→            <linearGradient id="langBar" x1="0" y1="0" x2="1" y2="0">
   608→              <stop offset="0%" stopColor="#7c3aed" />
   609→              <stop offset="100%" stopColor="#d946ef" />
   610→            </linearGradient>
   611→          </defs>
   612→          <CartesianGrid
   613→            strokeDasharray="3 3"
   614→            stroke="currentColor"
   615→            className="text-border/40"

   617→          />

   668→// ---------- Provider usage card ----------
   669→function ProviderUsageCard({
   670→  title,
   671→  icon: Icon,
   672→  accent,
   673→  data,
   674→  resolver,
   675→}: {
   676→  title: string;
   677→  icon: React.ComponentType<{ className?: string }>;
   678→  accent: "violet" | "fuchsia";
   679→  data: { id: string; count: number }[];
   680→  resolver: (id: string) => string;
   681→}) {
   682→  const total = data.reduce((a, b) => a + b.count, 0);
   683→  const accentClass =
   684→    accent === "violet"
   685→      ? "from-violet-500 to-fuchsia-500 text-violet-300"
   686→      : "from-fuchsia-500 to-pink-500 text-fuchsia-300";
   687→  return (
   688→    <Card className="glass card-glow h-full">
   689→      <CardHeader>
   690→        <div className="flex items-center gap-2">
   691→          <span
   692→            className={cn(
   693→              "grid size-8 place-items-center rounded-lg bg-gradient-to-br text-white shadow-md",
   694→              accentClass
   695→            )}
   696→          >
   697→            <Icon className="size-4" />
   698→          </span>
   699→          <div>
   700→            <CardTitle className="text-base">{title}</CardTitle>
   701→            <p className="text-xs text-muted-foreground">
   702→              Toplam {total} kullanım
   703→            </p>
   704→          </div>
   705→        </div>
   706→      </CardHeader>
   707→      <CardContent className="space-y-2">
   708→        {data.length === 0 ? (
   709→          <p className="text-sm text-muted-foreground py-4 text-center">
   710→            Henüz veri yok
   711→          </p>
   712→        ) : (
   713→          data.map((d, i) => {
   714→            const name = resolver(d.id);
   715→            const pct = total > 0 ? Math.round((d.count / total) * 100) : 0;
   716→            return (
   717→              <motion.div
   718→                key={d.id + i}
   719→                initial={{ opacity: 0, x: -8 }}
   720→                animate={{ opacity: 1, x: 0 }}
   721→                transition={{ delay: i * 0.04 }}
   722→                className="flex items-center gap-3"
   723→              >
   724→                <div className="min-w-0 flex-1">
   725→                  <div className="flex items-center justify-between text-sm mb-1">
   726→                    <span className="truncate font-medium">{name}</span>
   727→                    <span className="text-xs text-muted-foreground tabular-nums">
   728→                      {d.count} · %{pct}
   729→                    </span>
   730→                  </div>
   731→                  <div className="h-1.5 w-full rounded-full bg-muted overflow-hidden">
   732→                    <div
   733→                      className={cn(
   734→                        "h-full rounded-full bg-gradient-to-r",
   735→                        accentClass
   736→                      )}
   737→                      style={{ width: `${pct}%` }}
   738→                    />
   739→                  </div>
   740→                </div>
   741→                <Badge
   742→                  variant="outline"
   743→                  className={cn("shrink-0", accentClass)}
   744→                >
   745→                  {d.count}
   746→                </Badge>
   747→              </motion.div>
   748→            );
   749→          })
   750→        )}
   751→      </CardContent>
   752→    </Card>
   753→  );
   754→}
