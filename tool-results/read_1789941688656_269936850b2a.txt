   439→   439→function PromptSuggestions({
   440→   440→  mode,
   441→   441→  topic,
   442→   442→  language,
   443→   443→  onPick,
   444→   444→}: {
   445→   445→  mode: string;
   446→   446→  topic: string;
   447→   447→  language: string;
   448→   448→  onPick: (idea: string) => void;
   449→   449→}) {
   450→   450→  const [open, setOpen] = React.useState(false);
   451→   451→  const [curated, setCurated] = React.useState<string[]>([]);
   452→   452→  const [ai, setAi] = React.useState<string[]>([]);
   453→   453→  const [loadingCurated, setLoadingCurated] = React.useState(false);
   454→   454→  const [loadingAi, setLoadingAi] = React.useState(false);
   455→   455→  const [aiFetched, setAiFetched] = React.useState(false);
   456→   456→
   457→   457→  // Fetch curated ideas whenever the panel is opened (mode/language can change)
   458→   458→  React.useEffect(() => {
   459→   459→    if (!open) return;
   460→   460→    let alive = true;
   461→   461→    setLoadingCurated(true);
   462→   462→    setCurated([]);
   463→   463→    setAi([]);
   464→   464→    setAiFetched(false);
   465→   465→    suggestPrompts({ mode, language })
   466→   466→      .then((res) => {
   467→   467→        if (!alive) return;
   468→   468→        setCurated(res.curated || []);
   469→   469→      })
   470→   470→      .catch(() => {
   471→   471→        if (alive) setCurated([]);
   472→   472→      })
   473→   473→      .finally(() => {
   474→   474→        if (alive) setLoadingCurated(false);
   475→   475→      });
   476→   476→    return () => {
   477→   477→      alive = false;
   478→   478→    };
   479→   479→  }, [open, mode, language]);
   480→   480→
