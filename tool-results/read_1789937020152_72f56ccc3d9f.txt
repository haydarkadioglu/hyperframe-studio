   115→export function DetailView() {
   116→  const detailId = useApp((s) => s.detailId);
   117→  const go = useApp((s) => s.go);
   118→  const [project, setProject] = React.useState<VideoProject | null>(null);
   119→  const [loading, setLoading] = React.useState(true);
   120→  const [error, setError] = React.useState<string | null>(null);
   121→  const [pendingDelete, setPendingDelete] = React.useState(false);
   122→  const [rerendering, setRerendering] = React.useState(false);
   123→  const [duplicating, setDuplicating] = React.useState(false);
   124→  const [editMode, setEditMode] = React.useState(false);
   125→
   126→  // Poll when generating
   127→  React.useEffect(() => {
   128→    if (!detailId) {
   129→      setLoading(false);
   130→      return;
   131→    }
   132→    let alive = true;
   133→    let timer: ReturnType<typeof setTimeout> | null = null;
   134→
   135→    const fetchOnce = async () => {
   136→      try {
   137→        const p = await getProject(detailId);
   138→        if (!alive) return;
   139→        setProject(p);
   140→        setError(null);
   141→        if (p.status === "generating") {
   142→          timer = setTimeout(fetchOnce, 2500);
   143→        }
   144→      } catch (e: any) {
   145→        if (!alive) return;
   146→        setError(e?.message || "Proje yüklenemedi");
   147→      } finally {
   148→        if (alive) setLoading(false);
   149→      }
   150→    };
   151→
   152→    setLoading(true);
   153→    fetchOnce();
   154→
   155→    return () => {
   156→      alive = false;
   157→      if (timer) clearTimeout(timer);
   158→    };
   159→  }, [detailId]);
   160→
   161→  const handleRerender = async () => {
   162→    if (!project) return;
   163→    setRerendering(true);
   164→    try {
   165→      await renderProject(project.id, {
   166→        targetScenes: project.sceneCount || 5,
   167→      });
   168→      toast.success("Yeniden üretim başladı");
   169→      setProject({ ...project, status: "generating" });
   170→    } catch (e: any) {
   171→      toast.error("Yeniden üretim başlatılamadı", { description: e?.message });
   172→    } finally {
   173→      setRerendering(false);
   174→    }
   175→  };
   176→
   177→  const handleDelete = async () => {
   178→    if (!project) return;
   179→    try {
   180→      await deleteProject(project.id);
   181→      toast.success("Proje silindi");
   182→      go("projects");
   183→    } catch (e: any) {
   184→      toast.error("Silme hatası", { description: e?.message });
   185→    }
   186→  };
   187→
   188→  const handleDuplicate = async () => {
   189→    if (!project) return;
   190→    setDuplicating(true);
   191→    try {
   192→      const copy = await duplicateProject(project.id);
   193→      toast.success("Proje kopyalandı", {
   194→        description: "Yeni taslak oluşturuldu.",
   195→      });
   196→      if (copy.scenes && copy.scenes.length > 0) {
   197→        go("detail", copy.id);
   198→      } else {
   199→        go("projects");
   200→      }
   201→    } catch (e: any) {
   202→      toast.error("Kopyalama hatası", { description: e?.message });
   203→    } finally {
   204→      setDuplicating(false);
   205→    }
   206→  };
   207→
   208→  if (!detailId) {
   209→    return (
   210→      <Card className="border-dashed">
   211→        <CardContent className="py-12 text-center text-muted-foreground">
   212→          Geçersiz proje.
   213→        </CardContent>
   214→      </Card>
   215→    );
   216→  }
   217→
   218→  if (loading) {
   219→    return (
   220→      <div className="space-y-6">
   221→        <Skeleton className="h-9 w-32" />
   222→        <Skeleton className="h-8 w-2/3" />
   223→        <Skeleton className="aspect-video w-full rounded-2xl" />
   224→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   225→          {Array.from({ length: 4 }).map((_, i) => (
   226→            <Skeleton key={i} className="aspect-video rounded-lg" />
   227→          ))}
   228→        </div>
   229→      </div>
   230→    );
   231→  }
   232→
   233→  if (error || !project) {
   234→    return (
   235→      <Card className="border-dashed border-rose-500/40">
   236→        <CardContent className="py-12 text-center space-y-3">
   237→          <AlertCircle className="size-8 text-rose-500 mx-auto" />
   238→          <p className="font-medium">Proje yüklenemedi</p>
   239→          <p className="text-sm text-muted-foreground">{error ?? "Bilinmeyen hata"}</p>
   240→          <Button variant="outline" onClick={() => go("projects")}>
   241→            <ArrowLeft className="size-4" />
   242→            Projelere dön
   243→          </Button>
   244→        </CardContent>
   245→      </Card>
   246→    );
   247→  }
   248→
   249→  return (
   250→    <div className="space-y-6">
   251→      {/* Top nav */}
   252→      <div className="flex items-center justify-between gap-2">
   253→        <Button
   254→          variant="ghost"
   255→          size="sm"
   256→          onClick={() => go("projects")}
   257→          className="min-h-[40px]"
   258→        >
   259→          <ArrowLeft className="size-4" />
   260→          Projelere dön
   261→        </Button>
   262→        <div className="flex items-center gap-1 flex-wrap justify-end">
   263→          {project.status === "ready" && (
   264→            <Button
   265→              variant={editMode ? "default" : "outline"}
   266→              size="sm"
   267→              onClick={() => setEditMode((v) => !v)}
   268→              className={cn(
   269→                "min-h-[40px]",
   270→                editMode
   271→                  ? "bg-gradient-to-r from-violet-500 to-fuchsia-500 text-white border-0"
   272→                  : ""
   273→              )}
   274→            >
   275→              <PencilLine className="size-4" />
   276→              {editMode ? "Düzenlemeyi Bitir" : "Sahne Düzenle"}
   277→            </Button>
   278→          )}
   279→          <Button
   280→            variant="outline"
   281→            size="sm"
   282→            onClick={handleDuplicate}
   283→            disabled={duplicating}
   284→            className="min-h-[40px]"
   285→          >
   286→            {duplicating ? (
   287→              <Loader2 className="size-4 animate-spin" />
   288→            ) : (
   289→              <Copy className="size-4" />
   290→            )}
   291→            Kopyala
   292→          </Button>
   293→          <Button
   294→            variant="outline"
   295→            size="sm"
   296→            onClick={handleRerender}
   297→            disabled={rerendering || project.status === "generating"}
   298→            className="min-h-[40px]"
   299→          >
   300→            {rerendering ? (
   301→              <Loader2 className="size-4 animate-spin" />
   302→            ) : (
   303→              <RefreshCw className="size-4" />
   304→            )}
   305→            Yeniden Oluştur
   306→          </Button>
   307→          <Button
   308→            variant="outline"
   309→            size="sm"
   310→            onClick={() => setPendingDelete(true)}
   311→            className="text-rose-500 hover:text-rose-400 hover:bg-rose-500/10 border-rose-500/30 min-h-[40px]"
   312→          >
   313→            <Trash2 className="size-4" />
   314→            Sil
   315→          </Button>
   316→        </div>
   317→      </div>
   318→
   319→      {/* Title + meta */}
   320→      <div className="space-y-2">
   321→        <div className="flex flex-wrap items-center gap-2">
   322→          <Badge variant="outline" className="bg-fuchsia-500/10 text-fuchsia-300 border-fuchsia-500/30">
   323→            {MODE_MAP[project.mode]?.emoji} {MODE_MAP[project.mode]?.label}
   324→          </Badge>
   325→          <StatusBadge status={project.status} />
   326→          <span className="text-xs text-muted-foreground">
   327→            {timeAgo(project.createdAt)}
   328→          </span>
   329→        </div>
   330→        <h1 className="text-2xl sm:text-3xl font-bold tracking-tight">
   331→          {project.title}
   332→        </h1>
   333→        <p className="text-sm text-muted-foreground line-clamp-2">
   334→          {project.topic}
   335→        </p>
   336→      </div>
   337→
   338→      {/* Body */}
   339→      {project.status === "generating" ? (
   340→        <GeneratingView project={project} />
   341→      ) : project.status === "error" ? (
   342→        <ErrorView
   343→          message={project.errorMessage || "Bilinmeyen bir hata oluştu"}
   344→          onRetry={handleRerender}
   345→        />
   346→      ) : editMode ? (
   347→        <SceneEditor
   348→          project={project}
   349→          onSaved={(updated) => {
   350→            setProject(updated);
   351→            setEditMode(false);
   352→          }}
   353→          onCancel={() => setEditMode(false)}
   354→        />
   355→      ) : (
   356→        <ReadyView project={project} />
   357→      )}
   358→
   359→      <AlertDialog open={pendingDelete} onOpenChange={setPendingDelete}>
   360→        <AlertDialogContent>
   361→          <AlertDialogHeader>
   362→            <AlertDialogTitle>Projeyi sil?</AlertDialogTitle>
   363→            <AlertDialogDescription>
   364→              <span className="font-medium text-foreground">“{project.title}”</span>{" "}
   365→              kalıcı olarak silinecek. Bu işlem geri alınamaz.
   366→            </AlertDialogDescription>
   367→          </AlertDialogHeader>
   368→          <AlertDialogFooter>
   369→            <AlertDialogCancel>İptal</AlertDialogCancel>
   370→            <AlertDialogAction
   371→              onClick={(e) => {
   372→                e.preventDefault();
   373→                handleDelete();
   374→              }}
   375→              className="bg-rose-500 hover:bg-rose-600 text-white"
   376→            >
   377→              Sil
   378→            </AlertDialogAction>
   379→          </AlertDialogFooter>
   380→        </AlertDialogContent>
   381→      </AlertDialog>
   382→    </div>
   383→  );
   384→}

   569→function ReadyView({ project }: { project: VideoProject }) {
   570→  const modeInfo = MODE_MAP[project.mode];
   571→  const styleInfo = STYLE_MAP[project.style];
   572→  const langInfo = LANGUAGE_MAP[project.language];
   573→  const toneInfo = TONE_MAP[project.tone];
   574→
   575→  const [selectedScene, setSelectedScene] = React.useState(0);
   576→
   577→  const downloadSrt = () => {
   578→    if (!project.subtitles) {
   579→      toast.error("Altyazı bulunamadı");
   580→      return;
   581→    }
   582→    downloadTextFile(`${slug(project.title)}.srt`, project.subtitles, "text/plain");
   583→    toast.success("SRT indirildi");
   584→  };
   585→
   586→  const downloadVtt = () => {
   587→    if (!project.subtitles) {
   588→      toast.error("Altyazı bulunamadı");
   589→      return;
   590→    }
   591→    const vtt = "WEBVTT\n\n" + project.subtitles.replace(/(\d{2}:\d{2}:\d{2}),(\d{3})/g, "$1.$2");
   592→    downloadTextFile(`${slug(project.title)}.vtt`, vtt, "text/vtt");
   593→    toast.success("VTT indirildi");
   594→  };
   595→
   596→  const downloadAudio = () => {
   597→    if (!project.audioUrl) {
   598→      toast.error("Ses dosyası bulunamadı");
   599→      return;
   600→    }
   601→    downloadRemoteFile(project.audioUrl, `${slug(project.title)}.mp3`);
   602→    toast.success("Ses indirildi");

   735→function SceneEditor({
   736→  project,
   737→  onSaved,
   738→  onCancel,
   739→}: {
   740→  project: VideoProject;
   741→  onSaved: (updated: VideoProject) => void;
   742→  onCancel: () => void;
   743→}) {
   744→  const [localScenes, setLocalScenes] = React.useState<Scene[]>(() =>
   745→    project.scenes.map((s) => ({ ...s }))
   746→  );
   747→  const [originalScenes] = React.useState<Scene[]>(() =>
   748→    project.scenes.map((s) => ({ ...s }))
   749→  );
   750→  const [saving, setSaving] = React.useState(false);
   751→  const [confirmCancel, setConfirmCancel] = React.useState(false);
   752→  const [pendingDeleteIdx, setPendingDeleteIdx] = React.useState<number | null>(null);
   753→
   754→  const dirty = React.useMemo(
   755→    () => JSON.stringify(localScenes) !== JSON.stringify(originalScenes),
   756→    [localScenes, originalScenes]
   757→  );
   758→
   759→  // Real-time project preview: build a synthetic project so player reflects edits
   760→  const previewProject = React.useMemo<VideoProject>(
   761→    () => ({ ...project, scenes: localScenes }),
   762→    [project, localScenes]
   763→  );
   764→
   765→  const updateScene = (idx: number, patch: Partial<Scene>) => {
   766→    setLocalScenes((prev) =>
   767→      prev.map((s, i) => (i === idx ? { ...s, ...patch } : s))
   768→    );
   769→  };
   770→
   771→  const handleAddScene = () => {
   772→    const newIdx = localScenes.length;
   773→    const newScene: Scene = {
   774→      id: `scene-${Date.now()}-${newIdx}`,
   775→      index: newIdx,
   776→      type: "text",
   777→      title: "",
   778→      text: "Yeni sahne metni",
   779→      narration: "Yeni sahne seslendirmesi.",
   780→      subtitle: "Yeni sahne",
   781→      durationMs: 4000,
   782→      animation: "fade",
   783→      accentColor: "#d946ef",
   784→    };
   785→    setLocalScenes((prev) => [...prev, newScene]);
   786→    toast.success("Yeni sahne eklendi");
   787→  };
   788→
   789→  const handleDeleteScene = (idx: number) => {
   790→    setLocalScenes((prev) =>
   791→      prev
   792→        .filter((_, i) => i !== idx)
   793→        .map((s, i) => ({ ...s, index: i }))
   794→    );
   795→    setPendingDeleteIdx(null);
   796→    toast.success("Sahne silindi");
   797→  };
   798→
   799→  const handleSave = async () => {
   800→    setSaving(true);
   801→    try {
   802→      const updated = await updateProject(project.id, {
   803→        scenes: localScenes,
   804→      });
   805→      toast.success("Sahnelar kaydedildi ve altyazılar güncellendi", {
   806→        description: `${localScenes.length} sahne sunucuya gönderildi.`,
   807→      });
   808→      onSaved(updated);
   809→    } catch (e: any) {
   810→      toast.error("Kaydetme hatası", { description: e?.message });
   811→    } finally {
   812→      setSaving(false);
   813→    }
   814→  };
   815→
   816→  const handleCancel = () => {
   817→    if (dirty) {
   818→      setConfirmCancel(true);
   819→    } else {
   820→      onCancel();
   821→    }
   822→  };
   823→
   824→  return (
   825→    <div className="space-y-5">
   826→      {/* Live preview */}
   827→      <Card className="glass border-fuchsia-500/30">
   828→        <CardHeader>
   829→          <div className="flex items-center justify-between">
   830→            <div>
   831→              <CardTitle className="text-base flex items-center gap-2">
   832→                <PencilLine className="size-4 text-fuchsia-500" />
   833→                Sahne Düzenleyici
   834→              </CardTitle>
   835→              <CardDescription>
   836→                Değişiklikler canlı önizlemede görünür. Kaydedince altyazılar otomatik yeniden oluşturulur.
   837→              </CardDescription>
   838→            </div>
   839→            {dirty && (
   840→              <Badge className="bg-amber-500/15 text-amber-400 border-amber-500/30">
   841→                Kaydedilmemiş değişiklik
   842→              </Badge>
   843→            )}
   844→          </div>
   845→        </CardHeader>
   846→        <CardContent>
   847→          <div className="rounded-2xl p-[1.5px] bg-gradient-to-br from-violet-500/40 via-fuchsia-500/30 to-pink-500/40">
   848→            <div className="rounded-2xl overflow-hidden">
   849→              <ScenePlayer project={previewProject} />
   850→            </div>
   851→          </div>
   852→        </CardContent>
   853→      </Card>
   854→
   855→      {/* Scene cards */}
   856→      <div className="space-y-4">
   857→        {localScenes.map((scene, idx) => (
   858→          <SceneEditorCard
   859→            key={scene.id}
   860→            scene={scene}
   861→            idx={idx}
   862→            style={project.style}
   863→            aspect={project.aspectRatio}
   864→            projectId={project.id}
   865→            onChange={(patch) => updateScene(idx, patch)}
   866→            onDelete={() => setPendingDeleteIdx(idx)}
   867→          />
   868→        ))}
   869→      </div>
   870→
   871→      {/* Add scene */}
   872→      <Button
   873→        variant="outline"
   874→        onClick={handleAddScene}
   875→        className="w-full min-h-[44px] border-dashed border-fuchsia-500/40 text-fuchsia-500 hover:bg-fuchsia-500/5 hover:text-fuchsia-400"
   876→      >
   877→        <Plus className="size-4" />
   878→        Sahne Ekle
   879→      </Button>
   880→
   881→      {/* Sticky action bar */}
   882→      <div className="sticky bottom-4 z-30">
   883→        <Card className="glass-strong border-fuchsia-500/30 shadow-xl shadow-fuchsia-500/10">
   884→          <CardContent className="py-3 flex items-center justify-between gap-3">
   885→            <div className="text-xs text-muted-foreground">
   886→              <span className="font-medium text-foreground">{localScenes.length}</span> sahne ·
   887→              {dirty ? " kaydedilmedi" : " kaydedildi"}
   888→            </div>
   889→            <div className="flex items-center gap-2">
   890→              <Button
   891→                variant="ghost"
   892→                onClick={handleCancel}
   893→                disabled={saving}
   894→                className="min-h-[40px]"
   895→              >
   896→                <X className="size-4" />
   897→                İptal
   898→              </Button>
   899→              <Button
   900→                onClick={handleSave}
   901→                disabled={saving || !dirty}
   902→                className="bg-gradient-to-r from-violet-500 via-fuchsia-500 to-pink-500 text-white border-0 shadow-lg shadow-fuchsia-500/30 min-h-[40px] min-w-[180px]"
   903→              >
   904→                {saving ? (
   905→                  <Loader2 className="size-4 animate-spin" />
   906→                ) : (
   907→                  <Save className="size-4" />
   908→                )}
   909→                {saving ? "Kaydediliyor..." : "Değişiklikleri Kaydet"}
   910→              </Button>
   911→            </div>
   912→          </CardContent>
   913→        </Card>
   914→      </div>
   915→
   916→      {/* Confirm delete scene */}
   917→      <AlertDialog
   918→        open={pendingDeleteIdx !== null}
   919→        onOpenChange={(o) => !o && setPendingDeleteIdx(null)}
   920→      >
   921→        <AlertDialogContent>
   922→          <AlertDialogHeader>
   923→            <AlertDialogTitle>Sahneyi sil?</AlertDialogTitle>
   924→            <AlertDialogDescription>
   925→              <span className="font-medium text-foreground">
   926→                {pendingDeleteIdx !== null ? `${pendingDeleteIdx + 1}. sahne` : "Sahne"}
   927→              </span>{" "}
   928→              silinecek. Kalan sahneler otomatik yeniden numaralandırılır.
   929→            </AlertDialogDescription>
   930→          </AlertDialogHeader>
   931→          <AlertDialogFooter>
   932→            <AlertDialogCancel>İptal</AlertDialogCancel>
   933→            <AlertDialogAction
   934→              onClick={(e) => {
   935→                e.preventDefault();
   936→                if (pendingDeleteIdx !== null) handleDeleteScene(pendingDeleteIdx);
   937→              }}
   938→              className="bg-rose-500 hover:bg-rose-600 text-white"
   939→            >
   940→              <Trash2 className="size-4" />
   941→              Sil
   942→            </AlertDialogAction>
   943→          </AlertDialogFooter>
   944→        </AlertDialogContent>
   945→      </AlertDialog>
   946→
   947→      {/* Confirm cancel with unsaved */}
   948→      <AlertDialog open={confirmCancel} onOpenChange={setConfirmCancel}>
   949→        <AlertDialogContent>
   950→          <AlertDialogHeader>
   951→            <AlertDialogTitle>Kaydedilmemiş değişiklikler var</AlertDialogTitle>
   952→            <AlertDialogDescription>
   953→              Çıkarsan tüm düzenlemeler kaybolur. Devam edilsin mi?
   954→            </AlertDialogDescription>
   955→          </AlertDialogHeader>
   956→          <AlertDialogFooter>
   957→            <AlertDialogCancel>Vazgeç</AlertDialogCancel>
   958→            <AlertDialogAction
   959→              onClick={(e) => {
   960→                e.preventDefault();
   961→                setConfirmCancel(false);
   962→                onCancel();
   963→              }}
   964→              className="bg-rose-500 hover:bg-rose-600 text-white"
   965→            >
   966→              Yine de çık
   967→            </AlertDialogAction>
   968→          </AlertDialogFooter>
   969→        </AlertDialogContent>
   970→      </AlertDialog>
   971→    </div>
   972→  );
   973→}
