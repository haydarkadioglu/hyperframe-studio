   439→function PromptSuggestions({
   440→  mode,
   441→  topic,
   442→  language,
   443→  onPick,
   444→}: {
   445→  mode: string;
   446→  topic: string;
   447→  language: string;
   448→  onPick: (idea: string) => void;
   449→}) {
   450→  const [open, setOpen] = React.useState(false);
   451→  const [curated, setCurated] = React.useState<string[]>([]);
   452→  const [ai, setAi] = React.useState<string[]>([]);
   453→  const [loadingCurated, setLoadingCurated] = React.useState(false);
   454→  const [loadingAi, setLoadingAi] = React.useState(false);
   455→  const [aiFetched, setAiFetched] = React.useState(false);
   456→
   457→  // Fetch curated ideas whenever the panel is opened (mode/language can change)
   458→  React.useEffect(() => {
   459→    if (!open) return;
   460→    let alive = true;
   461→    setLoadingCurated(true);
   462→    setCurated([]);
   463→    setAi([]);
   464→    setAiFetched(false);
   465→    suggestPrompts({ mode, language })
   466→      .then((res) => {
   467→        if (!alive) return;
   468→        setCurated(res.curated || []);
   469→      })
   470→      .catch(() => {
   471→        if (alive) setCurated([]);
   472→      })
   473→      .finally(() => {
   474→        if (alive) setLoadingCurated(false);
   475→      });
   476→    return () => {
   477→      alive = false;
   478→    };
   479→  }, [open, mode, language]);
   480→
   481→  const fetchAi = React.useCallback(async () => {
   482→    if (topic.trim().length < 3) return;
   483→    setLoadingAi(true);
   484→    try {
   485→      const res = await suggestPrompts({ topic, mode, language });
   486→      setAi(res.ai || []);
   487→      setAiFetched(true);
   488→    } catch {
   489→      setAi([]);
   490→      setAiFetched(true);
   491→    } finally {
   492→      setLoadingAi(false);
   493→    }
   494→  }, [topic, mode, language]);
   495→
   496→  const canAi = topic.trim().length >= 3;
   497→
   498→  const pickIdea = (idea: string) => {
   499→    onPick(idea);
   500→    setOpen(false);
   501→  };
   502→
   503→  return (
   504→    <Collapsible open={open} onOpenChange={setOpen}>
   505→      <CollapsibleTrigger asChild>
   506→        <button
   507→          type="button"
   508→          className={cn(
   509→            "group flex items-center justify-between gap-2 w-full rounded-xl border border-violet-500/30 bg-violet-500/5 hover:bg-violet-500/10 px-3 py-2 text-sm transition-colors",
   510→            open && "bg-violet-500/10"
   511→          )}
   512→        >
   513→          <span className="flex items-center gap-2">
   514→            <span className="grid size-6 place-items-center rounded-md bg-violet-500/20 text-violet-500">
   515→              <Lightbulb className="size-3.5" />
   516→            </span>
   517→            <span className="font-medium">💡 Konu Önerileri</span>
   518→          </span>
   519→          <ChevronDown
   520→            className={cn(
   521→              "size-4 text-muted-foreground transition-transform duration-200",
   522→              open && "rotate-180"
   523→            )}
   524→          />
   525→        </button>
   526→      </CollapsibleTrigger>
   527→      <CollapsibleContent className="data-[state=open]:animate-in data-[state=open]:fade-in-0 data-[state=open]:slide-in-from-top-2">
   528→        <div className="mt-3 space-y-3">
   529→          {/* Curated ideas */}
   530→          <div className="space-y-2">
   531→            <p className="text-[11px] uppercase tracking-wider text-muted-foreground font-semibold">
   532→              Hazır Konular
   533→            </p>
   534→            {loadingCurated ? (
   535→              <div className="flex items-center gap-2 text-xs text-muted-foreground">
   536→                <Loader2 className="size-3 animate-spin" />
   537→                Yükleniyor...
   538→              </div>
   539→            ) : curated.length > 0 ? (
   540→              <div className="flex flex-wrap gap-1.5">
   541→                {curated.map((idea, i) => (
   542→                  <motion.button
   543→                    key={idea + i}
   544→                    initial={{ opacity: 0, y: 6, scale: 0.92 }}
   545→                    animate={{ opacity: 1, y: 0, scale: 1 }}
   546→                    transition={{ delay: i * 0.04 }}
   547→                    onClick={() => pickIdea(idea)}
   548→                    className="accent-bg-soft accent-text-soft accent-border hover:brightness-110 rounded-full px-3 py-1 text-xs transition-all text-left max-w-full badge-pop"
   549→                  >
   550→                    {idea}
   551→                  </motion.button>
   552→                ))}
   553→              </div>
   554→            ) : (
   555→              <p className="text-xs text-muted-foreground">Şu an için öneri yok.</p>
   556→            )}
   557→          </div>
   558→
   559→          {/* AI enhancement */}
   560→          <div className="space-y-2">
   561→            <div className="flex items-center justify-between gap-2">
   562→              <p className="text-[11px] uppercase tracking-wider text-muted-foreground font-semibold">
   563→                ✨ AI Önerileri
   564→              </p>
   565→              <Button
   566→                size="sm"
   567→                variant="ghost"
   568→                onClick={fetchAi}
   569→                disabled={!canAi || loadingAi}
   570→                className="h-7 px-2 text-xs btn-gradient text-white border-0 shine-on-hover relative overflow-hidden"
   571→              >
   572→                {loadingAi ? (
   573→                  <Loader2 className="size-3 animate-spin" />
   574→                ) : (
