   115→   115→export function DetailView() {
   116→   116→  const detailId = useApp((s) => s.detailId);
   117→   117→  const go = useApp((s) => s.go);
   118→   118→  const [project, setProject] = React.useState<VideoProject | null>(null);
   119→   119→  const [loading, setLoading] = React.useState(true);
   120→   120→  const [error, setError] = React.useState<string | null>(null);
   121→   121→  const [pendingDelete, setPendingDelete] = React.useState(false);
   122→   122→  const [rerendering, setRerendering] = React.useState(false);
   123→   123→  const [duplicating, setDuplicating] = React.useState(false);
   124→   124→  const [editMode, setEditMode] = React.useState(false);
   125→   125→
   126→   126→  // Poll when generating
   127→   127→  React.useEffect(() => {
   128→   128→    if (!detailId) {
   129→   129→      setLoading(false);
   130→   130→      return;
   131→   131→    }
   132→   132→    let alive = true;
   133→   133→    let timer: ReturnType<typeof setTimeout> | null = null;
   134→   134→
   135→   135→    const fetchOnce = async () => {
   136→   136→      try {
   137→   137→        const p = await getProject(detailId);
   138→   138→        if (!alive) return;
   139→   139→        setProject(p);
   140→   140→        setError(null);
   141→   141→        if (p.status === "generating") {
   142→   142→          timer = setTimeout(fetchOnce, 2500);
   143→   143→        }
   144→   144→      } catch (e: any) {
   145→   145→        if (!alive) return;
   146→   146→        setError(e?.message || "Proje yüklenemedi");
   147→   147→      } finally {
   148→   148→        if (alive) setLoading(false);
   149→   149→      }
   150→   150→    };
   151→   151→
   152→   152→    setLoading(true);
   153→   153→    fetchOnce();
   154→   154→
   155→   155→    return () => {
   156→   156→      alive = false;
   157→   157→      if (timer) clearTimeout(timer);
   158→   158→    };
   159→   159→  }, [detailId]);
   160→   160→
   161→   161→  const handleRerender = async () => {
   162→   162→    if (!project) return;
   163→   163→    setRerendering(true);
   164→   164→    try {
   165→   165→      await renderProject(project.id, {
   166→   166→        targetScenes: project.sceneCount || 5,
   167→   167→      });
   168→   168→      toast.success("Yeniden üretim başladı");
   169→   169→      setProject({ ...project, status: "generating" });
   170→   170→    } catch (e: any) {
   171→   171→      toast.error("Yeniden üretim başlatılamadı", { description: e?.message });
   172→   172→    } finally {
   173→   173→      setRerendering(false);
   174→   174→    }
   175→   175→  };
   176→   176→
   177→   177→  const handleDelete = async () => {
   178→   178→    if (!project) return;
   179→   179→    try {
   180→   180→      await deleteProject(project.id);
   181→   181→      toast.success("Proje silindi");
   182→   182→      go("projects");
   183→   183→    } catch (e: any) {
   184→   184→      toast.error("Silme hatası", { description: e?.message });
   185→   185→    }
   186→   186→  };
   187→   187→
   188→   188→  const handleDuplicate = async () => {
   189→   189→    if (!project) return;
   190→   190→    setDuplicating(true);
   191→   191→    try {
   192→   192→      const copy = await duplicateProject(project.id);
   193→   193→      toast.success("Proje kopyalandı", {
   194→   194→        description: "Yeni taslak oluşturuldu.",
   195→   195→      });
   196→   196→      if (copy.scenes && copy.scenes.length > 0) {
   197→   197→        go("detail", copy.id);
   198→   198→      } else {
   199→   199→        go("projects");
   200→   200→      }
   201→   201→    } catch (e: any) {
   202→   202→      toast.error("Kopyalama hatası", { description: e?.message });
   203→   203→    } finally {
   204→   204→      setDuplicating(false);
   205→   205→    }
   206→   206→  };
   207→   207→
   208→   208→  if (!detailId) {
   209→   209→    return (
   210→   210→      <Card className="border-dashed">
   211→   211→        <CardContent className="py-12 text-center text-muted-foreground">
   212→   212→          Geçersiz proje.
   213→   213→        </CardContent>
   214→   214→      </Card>
   215→   215→    );
   216→   216→  }
   217→   217→
   218→   218→  if (loading) {
   219→   219→    return (
   220→   220→      <div className="space-y-6">
   221→   221→        <Skeleton className="h-9 w-32" />
   222→   222→        <Skeleton className="h-8 w-2/3" />
   223→   223→        <Skeleton className="aspect-video w-full rounded-2xl" />
   224→   224→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   225→   225→          {Array.from({ length: 4 }).map((_, i) => (
   226→   226→            <Skeleton key={i} className="aspect-video rounded-lg" />
   227→   227→          ))}
   228→   228→        </div>
   229→   229→      </div>
   230→   230→    );
   231→   231→  }
   232→   232→
   233→   233→  if (error || !project) {
   234→   234→    return (
   235→   235→      <Card className="border-dashed border-rose-500/40">
   236→   236→        <CardContent className="py-12 text-center space-y-3">
   237→   237→          <AlertCircle className="size-8 text-rose-500 mx-auto" />
   238→   238→          <p className="font-medium">Proje yüklenemedi</p>
   239→   239→          <p className="text-sm text-muted-foreground">{error ?? "Bilinmeyen hata"}</p>
   240→   240→          <Button variant="outline" onClick={() => go("projects")}>
   241→   241→            <ArrowLeft className="size-4" />
   242→   242→            Projelere dön
   243→   243→          </Button>
   244→   244→        </CardContent>
   245→   245→      </Card>
   246→   246→    );
   247→   247→  }
   248→   248→
   249→   249→  return (
   250→   250→    <div className="space-y-6">
   251→   251→      {/* Top nav */}
   252→   252→      <div className="flex items-center justify-between gap-2">
   253→   253→        <Button
   254→   254→          variant="ghost"
   255→   255→          size="sm"
   256→   256→          onClick={() => go("projects")}
   257→   257→          className="min-h-[40px]"
   258→   258→        >
   259→   259→          <ArrowLeft className="size-4" />
   260→   260→          Projelere dön
   261→   261→        </Button>
   262→   262→        <div className="flex items-center gap-1 flex-wrap justify-end">
   263→   263→          {project.status === "ready" && (
   264→   264→            <Button
   265→   265→              variant={editMode ? "default" : "outline"}
   266→   266→              size="sm"
   267→   267→              onClick={() => setEditMode((v) => !v)}
   268→   268→              className={cn(
   269→   269→                "min-h-[40px]",
   270→   270→                editMode
   271→   271→                  ? "bg-gradient-to-r from-violet-500 to-fuchsia-500 text-white border-0"
   272→   272→                  : ""
   273→   273→              )}
   274→   274→            >
   275→   275→              <PencilLine className="size-4" />
   276→   276→              {editMode ? "Düzenlemeyi Bitir" : "Sahne Düzenle"}
   277→   277→            </Button>
   278→   278→          )}
   279→   279→          <Button
   280→   280→            variant="outline"
   281→   281→            size="sm"
   282→   282→            onClick={handleDuplicate}
   283→   283→            disabled={duplicating}
   284→   284→            className="min-h-[40px]"
   285→   285→          >
   286→   286→            {duplicating ? (
   287→   287→              <Loader2 className="size-4 animate-spin" />
   288→   288→            ) : (
   289→   289→              <Copy className="size-4" />
   290→   290→            )}
   291→   291→            Kopyala
   292→   292→          </Button>
   293→   293→          <Button
   294→   294→            variant="outline"
   295→   295→            size="sm"
   296→   296→            onClick={handleRerender}
   297→   297→            disabled={rerendering || project.status === "generating"}
   298→   298→            className="min-h-[40px]"
   299→   299→          >
   300→   300→            {rerendering ? (
   301→   301→              <Loader2 className="size-4 animate-spin" />
   302→   302→            ) : (
   303→   303→              <RefreshCw className="size-4" />
   304→   304→            )}
   305→   305→            Yeniden Oluştur
   306→   306→          </Button>
   307→   307→          <Button
   308→   308→            variant="outline"
   309→   309→            size="sm"
   310→   310→            onClick={() => setPendingDelete(true)}
   311→   311→            className="text-rose-500 hover:text-rose-400 hover:bg-rose-500/10 border-rose-500/30 min-h-[40px]"
   312→   312→          >
   313→   313→            <Trash2 className="size-4" />
   314→   314→            Sil
   315→   315→          </Button>
   316→   316→        </div>
   317→   317→      </div>
   318→   318→
   319→   319→      {/* Title + meta */}
   320→   320→      <div className="space-y-2">
   321→   321→        <div className="flex flex-wrap items-center gap-2">
   322→   322→          <Badge variant="outline" className="bg-fuchsia-500/10 text-fuchsia-300 border-fuchsia-500/30">
   323→   323→            {MODE_MAP[project.mode]?.emoji} {MODE_MAP[project.mode]?.label}
   324→   324→          </Badge>
   325→   325→          <StatusBadge status={project.status} />
   326→   326→          <span className="text-xs text-muted-foreground">
   327→   327→            {timeAgo(project.createdAt)}
   328→   328→          </span>
   329→   329→        </div>
   330→   330→        <h1 className="text-2xl sm:text-3xl font-bold tracking-tight">
   331→   331→          {project.title}
   332→   332→        </h1>
   333→   333→        <p className="text-sm text-muted-foreground line-clamp-2">
   334→   334→          {project.topic}
   335→   335→        </p>
   336→   336→      </div>
   337→   337→
   338→   338→      {/* Body */}
   339→   339→      {project.status === "generating" ? (
   340→   340→        <GeneratingView project={project} />
   341→   341→      ) : project.status === "error" ? (
   342→   342→        <ErrorView
   343→   343→          message={project.errorMessage || "Bilinmeyen bir hata oluştu"}
   344→   344→          onRetry={handleRerender}
   345→   345→        />
   346→   346→      ) : editMode ? (
   347→   347→        <SceneEditor
   348→   348→          project={project}
   349→   349→          onSaved={(updated) => {
   350→   350→            setProject(updated);
   351→   351→            setEditMode(false);
   352→   352→          }}
   353→   353→          onCancel={() => setEditMode(false)}
   354→   354→        />
   355→   355→      ) : (
   356→   356→        <ReadyView project={project} />
   357→   357→      )}
   358→   358→
   359→   359→      <AlertDialog open={pendingDelete} onOpenChange={setPendingDelete}>
   360→   360→        <AlertDialogContent>
   361→   361→          <AlertDialogHeader>
   362→   362→            <AlertDialogTitle>Projeyi sil?</AlertDialogTitle>
   363→   363→            <AlertDialogDescription>
   364→   364→              <span className="font-medium text-foreground">“{project.title}”</span>{" "}
   365→   365→              kalıcı olarak silinecek. Bu işlem geri alınamaz.
   366→   366→            </AlertDialogDescription>
   367→   367→          </AlertDialogHeader>
   368→   368→          <AlertDialogFooter>
   369→   369→            <AlertDialogCancel>İptal</AlertDialogCancel>
   370→   370→            <AlertDialogAction
   371→   371→              onClick={(e) => {
   372→   372→                e.preventDefault();
   373→   373→                handleDelete();
   374→   374→              }}
   375→   375→              className="bg-rose-500 hover:bg-rose-600 text-white"
   376→   376→            >
   377→   377→              Sil
   378→   378→            </AlertDialogAction>
   379→   379→          </AlertDialogFooter>
   380→   380→        </AlertDialogContent>
   381→   381→      </AlertDialog>
   382→   382→    </div>
   383→   383→  );
   384→   384→}

   735→   735→function SceneEditor({

   738→   738→  onCancel,
   739→   739→}: {
   740→   740→  project: VideoProject;
   741→   741→  onSaved: (updated: VideoProject) => void;
   742→   742→  onCancel: () => void;
   743→   743→}) {
   744→   744→  const [localScenes, setLocalScenes] = React.useState<Scene[]>(() =>
   745→   745→    project.scenes.map((s) => ({ ...s }))
   746→   746→  );
   747→   747→  const [originalScenes] = React.useState<Scene[]>(() =>
   748→   748→    project.scenes.map((s) => ({ ...s }))
   749→   749→  );
   750→   750→  const [saving, setSaving] = React.useState(false);
   751→   751→  const [confirmCancel, setConfirmCancel] = React.useState(false);
   752→   752→  const [pendingDeleteIdx, setPendingDeleteIdx] = React.useState<number | null>(null);
   753→   753→
   754→   754→  const dirty = React.useMemo(
   755→   755→    () => JSON.stringify(localScenes) !== JSON.stringify(originalScenes),
   756→   756→    [localScenes, originalScenes]
   757→   757→  );
   758→   758→
   759→   759→  // Real-time project preview: build a synthetic project so player reflects edits
   760→   760→  const previewProject = React.useMemo<VideoProject>(
   761→   761→    () => ({ ...project, scenes: localScenes }),
   762→   762→    [project, localScenes]
   763→   763→  );
   764→   764→
   765→   765→  const updateScene = (idx: number, patch: Partial<Scene>) => {
   766→   766→    setLocalScenes((prev) =>
   767→   767→      prev.map((s, i) => (i === idx ? { ...s, ...patch } : s))
   768→   768→    );
   769→   769→  };
   770→   770→
   771→   771→  const handleAddScene = () => {
   772→   772→    const newIdx = localScenes.length;
   773→   773→    const newScene: Scene = {
   774→   774→      id: `scene-${Date.now()}-${newIdx}`,
   775→   775→      index: newIdx,
   776→   776→      type: "text",
   777→   777→      title: "",
   778→   778→      text: "Yeni sahne metni",
   779→   779→      narration: "Yeni sahne seslendirmesi.",
   780→   780→      subtitle: "Yeni sahne",
   781→   781→      durationMs: 4000,
   782→   782→      animation: "fade",
   783→   783→      accentColor: "#d946ef",
   784→   784→    };

   786→   786→    toast.success("Yeni sahne eklendi");
   787→   787→  };
   788→   788→
   789→   789→  const handleDeleteScene = (idx: number) => {
   790→   790→    setLocalScenes((prev) =>
   791→   791→      prev
   792→   792→        .filter((_, i) => i !== idx)
   793→   793→        .map((s, i) => ({ ...s, index: i }))
   794→   794→    );
   795→   795→    setPendingDeleteIdx(null);
   796→   796→    toast.success("Sahne silindi");
   797→   797→  };
   798→   798→
   799→   799→  const handleSave = async () => {
   800→   800→    setSaving(true);

   802→   802→      const updated = await updateProject(project.id, {
   803→   803→        scenes: localScenes,
   804→   804→      });
   805→   805→      toast.success("Sahnelar kaydedildi ve altyazılar güncellendi", {
   806→   806→        description: `${localScenes.length} sahne sunucuya gönderildi.`,
   807→   807→      });
   808→   808→      onSaved(updated);
   809→   809→    } catch (e: any) {
   810→   810→      toast.error("Kaydetme hatası", { description: e?.message });
   811→   811→    } finally {
   812→   812→      setSaving(false);
   813→   813→    }
   814→   814→  };
   815→   815→
   816→   816→  const handleCancel = () => {

   818→   818→      setConfirmCancel(true);
   819→   819→    } else {
   820→   820→      onCancel();
   821→   821→    }
   822→   822→  };
   823→   823→
   824→   824→  return (
   825→   825→    <div className="space-y-5">

   829→   829→          <div className="flex items-center justify-between">
   830→   830→            <div>
   831→   831→              <CardTitle className="text-base flex items-center gap-2">
   832→   832→                <PencilLine className="size-4 text-fuchsia-500" />
   833→   833→                Sahne Düzenleyici

   840→   840→              <Badge className="bg-amber-500/15 text-amber-400 border-amber-500/30">
   841→   841→                Kaydedilmemiş değişiklik
   842→   842→              </Badge>
   843→   843→            )}

   846→   846→        <CardContent>
